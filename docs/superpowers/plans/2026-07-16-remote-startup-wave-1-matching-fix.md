```ts
export function roleFuzzyMatch(target: RoleTarget, posting: RawPosting): boolean {
```

```ts
// Coarse job-function classifier for stage-1 tagging (design spine §2.1/§2.2,
// spec 2026-07-16-remote-startup-niche-source-expansion-design.md §4.1
// "Function tagging"). Pure, deterministic, best-effort search-input
// classification — NOT a wire/contract boundary: an unclassifiable title
// bucketing to "other" is a real modeled class, not a fail-loud violation.
// Wave 3's LLM classifier (deferred) refines only the survivors this tags
// "other" or ambiguous, at pool scale; this file never calls an LLM.
//
// Deliberately independent of roleMatch.ts's tokenizer: that tokenizer drops
// "head"/"chief" as ROLE_STOPWORDS (seniority words, not matching signal),
// but classifyFunction needs exactly those words as its exec-level signal —
// reusing roleTokens() would silently break exec detection.
export type JobFunction =
  | "eng"
  | "product"
  | "design"
  | "ops"
  | "finance"
  | "legal"
  | "marketing"
  | "sales"
  | "people"
  | "cs"
  | "exec"
  | "other";

function tokenize(title: string): Set<string> {
  return new Set(
    title
      .toLowerCase()
      .replace(/[^a-z0-9&]+/g, " ")
      .trim()
      .split(/\s+/)
      .filter(Boolean),
  );
}

// Exec is a level, not a function — checked before the function buckets, so
// "Head of Finance"/"VP of Marketing" tag exec (matching the donor exec
// titles, spec §3), not the function word they also carry.
const EXEC_TOKENS = new Set(["ceo", "cfo", "coo", "cto", "cmo", "cro", "cpo", "vp", "chief", "head"]);

// Checked in this order after the exec gate — first bucket whose keyword(s)
// are present wins. Atoms are the design-spine §2.2 list verbatim.
// "account executive"/"customer success" are 2-word phrases, checked
// against the raw lowercased title (a token Set doesn't preserve adjacency).
const FUNCTION_RULES: [JobFunction, (tokens: Set<string>, lower: string) => boolean][] = [
  ["finance", (t) => t.has("finance") || t.has("controller") || t.has("accounting") || t.has("fp&a")],
  ["legal", (t) => t.has("legal") || t.has("counsel") || t.has("paralegal")],
  ["people", (t) => t.has("recruiter") || t.has("talent") || t.has("people") || t.has("hr")],
  ["sales", (t, lower) => t.has("sales") || t.has("sdr") || lower.includes("account executive")],
  ["marketing", (t) => t.has("marketing") || t.has("growth") || t.has("brand")],
  ["cs", (t, lower) => t.has("support") || t.has("csm") || lower.includes("customer success")],
  ["ops", (t) => t.has("ops") || t.has("operations")],
  ["product", (t) => t.has("product") || t.has("pm")],
  ["design", (t) => t.has("design") || t.has("ux") || t.has("ui")],
  ["eng", (t) => t.has("engineer") || t.has("developer") || t.has("data") || t.has("sre")],
];

export function classifyFunction(title: string): JobFunction {
  const lower = title.toLowerCase();
  const tokens = tokenize(title);
  if ([...EXEC_TOKENS].some((t) => tokens.has(t))) return "exec";
  for (const [fn, matches] of FUNCTION_RULES) {
    if (matches(tokens, lower)) return fn;
  }
  return "other";
}
```

```ts
/**
 * Deterministic stage-1 ranking score (design spine §2.6) — NOT a gate (that
 * remains roleFuzzyMatch/the run.ts:308 check); purely for ordering already-
 * admitted candidates before the top-N slice. Exact-title match (design
 * spine §2.3) ranks highest (identicals first); otherwise the best score
 * across target.titles using the same overlap/discriminating/Jaccard shape
 * as titleMatchesPosting, computed unconditionally (no floor/containment
 * short-circuits — those are match/no-match rules, irrelevant to ranking
 * postings that already passed the gate).
 */
export function roleMatchScore(target: RoleTarget, posting: RawPosting): number {
  const normalizedPostingTitle = normalizeTitle(posting.title);
  if (target.titles.some((t) => isExactTitleMatch(t, normalizedPostingTitle))) return 1000;

  const postingTokens = [...new Set(roleTokens(posting.title))];
  if (postingTokens.length === 0) return 0;
  const postingSet = new Set(postingTokens);
  const keywordTokens = new Set(target.keywords.flatMap((keyword) => roleTokens(keyword)));

  let best = 0;
  for (const title of target.titles) {
    const titleTokens = [...new Set(roleTokens(title))];
    if (titleTokens.length === 0) continue;
    const titleTokenSet = new Set(titleTokens);
    const titleOverlap = titleTokens.filter((w) => postingSet.has(w));
    const keywordOverlap = [...keywordTokens].filter((w) => !titleTokenSet.has(w) && postingSet.has(w));
    const overlap = [...titleOverlap, ...keywordOverlap];
    const discriminating = overlap.filter((w) => !BASELINE_TOKENS.has(w));
    const union = new Set([...titleTokens, ...postingTokens]).size;
    const jaccard = union === 0 ? 0 : overlap.length / union;
    const score = discriminating.length * 10 + Math.round(jaccard * 10);
    best = Math.max(best, score);
  }
  return best;
}
```

```ts
function groupByCollision(matched: { posting: RawPosting; source: SourceRow; stage1Score: number }[]): Map<string, CanonicalGroup> {
  const groups = new Map<string, CanonicalGroup>();
  for (const { posting, source, stage1Score } of matched) {
    const key = secondaryKey({
      companySlug: companySlugFor(posting.company),
      roleTokensHash: roleTokensHash(posting.title),
      location: posting.location ?? "",
    });

    const existing = groups.get(key);
    if (!existing) {
      groups.set(key, { canonical: posting, canonicalSource: source, aliasUrls: [], stage1Score });
      continue;
    }

    // Keep the max stage1Score across a collision (design-spine §2.6) —
    // whichever duplicate ranked higher carries the group forward,
    // regardless of which posting wins the canonical URL below.
    existing.stage1Score = Math.max(existing.stage1Score, stage1Score);

    const resolved = resolveCanonicalCollision(
      { kind: existing.canonicalSource.kind, sourceId: existing.canonicalSource.id, url: existing.canonical.url },
      { kind: source.kind, sourceId: source.id, url: posting.url },
    );
    if (resolved.canonical.url === posting.url) {
      existing.aliasUrls.push({ sourceId: existing.canonicalSource.id, url: existing.canonical.url });
      existing.canonical = posting;
      existing.canonicalSource = source;
    } else {
      existing.aliasUrls.push({ sourceId: source.id, url: posting.url });
    }
  }
  return groups;
}
```

```ts
// bug): Map-insertion/network-race order must never reach the .slice() in
// scoreTopCandidates. Exported so this test can assert the ordering is
// stable under input reordering without depending on real connector-fan-out

export function sortCandidatesForScoring<T extends { job: Pick<JobRow, "postedAt" | "dedupeKey">; stage1Score: number }>(
  candidates: T[],
): T[] {
  return [...candidates].sort((a, b) => {
    if (b.stage1Score !== a.stage1Score) return b.stage1Score - a.stage1Score;
    const aPosted = a.job.postedAt?.getTime() ?? -Infinity;

    if (bPosted !== aPosted) return bPosted - aPosted;
    return a.job.dedupeKey < b.job.dedupeKey ? -1 : a.job.dedupeKey > b.job.dedupeKey ? 1 : 0;
  });
}
```
